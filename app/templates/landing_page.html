Quickstart

From zero to first API call in 30 seconds.

# Step 1: Register for a free API key
curl -X POST http://178.105.35.170:8080/register \
  -H "Content-Type: application/json" \
  -d '{"email": "you@example.com"}'

# Response: {"success": true, "api_key": "hk_live_...", "credits": 10}

# Step 2: Summarize text
curl -X POST http://178.105.35.170:8080/v1/execute \
  -H "Content-Type: application/json" \
  -H "X-API-Key: hk_live_YOUR_KEY" \
  -d '{
    "task": "summarize",
    "text": "Your long text here..."
  }'

# Step 3: Extract structured JSON
curl -X POST http://178.105.35.170:8080/v1/execute \
  -H "X-API-Key: hk_live_YOUR_KEY" \
  -d '{
    "task": "extract_json",
    "text": "Meeting Jan 15 2026 at 3pm with John about Q1 budget"
  }'
